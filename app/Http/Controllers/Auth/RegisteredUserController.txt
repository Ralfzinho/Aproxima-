<?php

namespace App\Http\Controllers\Auth;

use App\Http\Controllers\Controller;
use App\Models\User;
use App\Models\Causa;
use Illuminate\Http\Request;
use Illuminate\Validation\Rule;
use Illuminate\Validation\Rules\Password;
use Illuminate\Support\Facades\Auth;

class RegisteredUserController extends Controller
{
     // Mostra a tela de cadastro passando as causas do banco
    public function create()
    {
        $causas = Causa::orderBy('nome')->get(['id', 'nome']);
        return view('cadastro_voluntario',compact('causas'));
    }
    
    public function store(Request $request)
    {
            $data = $request->validate([
            'name'            => ['required','string','max:150'],
            'email'           => ['required','string','lowercase','email','max:150','unique:users,email'],
            'password'        => ['required','confirmed', Password::defaults()],

            'telefone'        => ['nullable','string','max:30'],
            'estado'          => ['nullable','string','size:2'],
            'cidade'          => ['nullable','string','max:100'],
            'dta_nascimento'  => ['nullable','date','before:today'],

            'profissao'       => ['nullable','string','max:120'],
            'habilidades'     => ['nullable','string','max:2000'],

            'disponibilidade' => ['nullable', Rule::in(array_keys(User::DISPONIBILIDADES))],
            'experiencia'     => ['nullable', Rule::in(array_keys(User::EXPERIENCIAS))],
            'tempo_semana'    => ['nullable', Rule::in(array_keys(User::TEMPOS_SEMANA))],

        ]);

        $user = User::create([
            'name'            => $data['name'],
            'email'           => $data['email'],
            'password'        => $data['password'], // hash via cast do Model
            'telefone'        => $data['telefone'] ?? null,
            'estado'          => $data['estado'] ?? null,
            'cidade'          => $data['cidade'] ?? null,
            'dta_nascimento'  => $data['dta_nascimento'] ?? null,
            'profissao'       => $data['profissao'] ?? null,
            'habilidades'     => $data['habilidades'] ?? null,
            'disponibilidade' => $data['disponibilidade'] ?? null,
            'experiencia'     => $data['experiencia'] ?? null,
            'tempo_semana'    => $data['tempo_semana'] ?? null,
        ]); $user->causas()->sync($data['causas'] ?? []);

        Auth::login($user);
        $request->session()->regenerate();

        return redirect()->intended(route('dashboard'));
    }
}
